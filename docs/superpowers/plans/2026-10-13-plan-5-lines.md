# Lines Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let users draw, view, edit, hide and delete named military control lines (10 types) made of two or more points, by tapping the map or typing MGRS grids.

**Architecture:** A line is a new map item kind (`kind = 'line'`, GeoJSON LineString, line type in `style`). Pure modules hold the line types, geometry, measurement and the vertex-editing rules; the draft store carries vertices + an undo stack; `LinesLayer` draws saved lines and `DraftLineLayer` the one being edited. Add/Edit menus reuse the item panels through a per-category adapter, and the map screen routes taps to the line draft while drawing.

**Tech Stack:** Expo SDK 57, React Native 0.86, WatermelonDB 0.28, @rnmapbox/maps 10, Zustand, zod, @resvg/resvg-js (image build), Jest + RNTL 14.

**Spec:** `docs/superpowers/specs/2026-10-13-lines-design.md`

## Global Constraints

- No schema change: `kind = 'line'`, LineString in `geometry`, `{"lineType": …}` in `style`, `sidc` null.
- 10 line types exactly as spec §3.1 (ids, labels, prefixes, dash, width, arrow, decoration).
- All lines drawn in the `friendly` color with the `point-symbol-halo` halo. Widths (pt): normal 3, heavy 5, wide 7; halo 2 pt wider; dash `[2, 1.5]`.
- A line needs a non-blank name and ≥ 2 points to save. No auto-names for lines.
- `itemCategory({ kind, sidc })` is the only unit/point/line decision.
- Toolbar **UNIT · POINT · LINE · LAYERS**; LINE icon `vector-polyline`.
- Menu ids `line.add` `{}` and `line.edit` `{ itemId }`; titles "Add line" / "Line" / "Edit line".
- Undo history max 50; line type changes are not undoable.
- Out of scope: polygons, measure, freehand, dragging markers, paste a list, curves, per-type colors, boundary echelons, repeated labels.
- Repo rules: `npx jest` green, `npx tsc --noEmit` clean, `npx expo lint` 0 problems; single quotes; never run prettier; Conventional Commits; **no Co-Authored-By or any other trailer in commit messages**; no hex colors outside `src/theme` (scripts may read `mapColors`).

## Review Focus

1. Removing points from a line that has exactly 2 must be blocked (× disabled + hint), but while first building a line (0–1 points) removal is free — Task 3 (`line-draft` tests) and Task 7 (menu test).
2. A typed grid that is partial or invalid must never add a point; return in the last field must add a valid one — Task 7.
3. While drawing, tapping an existing unit/point symbol adds a line point there instead of opening that item — Task 8 (MapScreen test).
4. Undo after insert-after and after remove restores the exact previous vertices and selection — Task 3.
5. A saved line whose stored geometry is unreadable (bad JSON, < 2 positions) is skipped on the map and opening it reports "That line no longer exists." — Task 2 (selector test) and Task 7 (menu test).

```ts
/** Military control lines the Line menu offers (spec 2026-10-13-lines-design.md §3.1). */
export type LineType =
  | 'phase-line'
  | 'line-of-departure'
  | 'limit-of-advance'
  | 'boundary'
  | 'route'
  | 'axis-of-advance'
  | 'flot'
  | 'restrictive-fire-line'
  | 'coordinated-fire-line'
  | 'fscl';

export type LineTypeInfo = {
  id: LineType;
  label: string;
  /** Label prefix ("PL"); null shows the name alone (boundary). */
  prefix: string | null;
  dash: 'solid' | 'dashed';
  width: 'normal' | 'heavy' | 'wide';
  endArrow: boolean;
  decoration: 'none' | 'flot-arcs';
};

const t = (id: LineType, label: string, prefix: string | null, extra: Partial<LineTypeInfo> = {}): LineTypeInfo => ({
  id,
  label,
  prefix,
  dash: 'solid',
  width: 'normal',
  endArrow: false,
  decoration: 'none',
  ...extra,
});

export const LINE_TYPES: readonly LineTypeInfo[] = [
  t('phase-line', 'Phase line', 'PL'),
  t('line-of-departure', 'Line of departure', 'LD'),
  t('limit-of-advance', 'Limit of advance', 'LOA'),
  t('boundary', 'Boundary', null, { width: 'heavy' }),
  t('route', 'Route', 'RTE', { endArrow: true }),
  t('axis-of-advance', 'Axis of advance', 'AXIS', { width: 'wide', endArrow: true }),
  t('flot', 'Forward line of own troops', 'FLOT', { decoration: 'flot-arcs' }),
  t('restrictive-fire-line', 'Restrictive fire line', 'RFL', { dash: 'dashed' }),
  t('coordinated-fire-line', 'Coordinated fire line', 'CFL', { dash: 'dashed' }),
  t('fscl', 'Fire support coordination line', 'FSCL'),
];

const BY_ID = new Map(LINE_TYPES.map((x) => [x.id, x]));

export function lineTypeInfo(id: LineType): LineTypeInfo {
  return BY_ID.get(id)!;
}

/** "PL RED"; the prefix is not repeated when the name already starts with it; boundaries show the name. */
export function lineLabel(type: LineType, name: string): string {
  const trimmed = name.trim();
  const prefix = lineTypeInfo(type).prefix;
  if (!prefix || trimmed.startsWith(`${prefix} `) || trimmed === prefix) return trimmed;
  return `${prefix} ${trimmed}`;
}

export function lineStyle(type: LineType): string {
  return JSON.stringify({ lineType: type });
}

/** The line type stored in an item's `style`; unknown or missing values read as a phase line. */
export function lineTypeOf(style: string | null): LineType {
  if (!style) return 'phase-line';
  try {
    const value = (JSON.parse(style) as { lineType?: unknown }).lineType;
    return typeof value === 'string' && BY_ID.has(value as LineType) ? (value as LineType) : 'phase-line';
  } catch {
    return 'phase-line';
  }
}
```

In `src/features/map-items/geometry.ts` add:

```ts
export type LineGeometry = { type: 'LineString'; coordinates: [number, number][] };

export function lineGeometry(points: LatLon[]): string {
  return JSON.stringify({ type: 'LineString', coordinates: points.map((p) => [p.lon, p.lat]) });
}

/** The line's points, or null for bad JSON, another geometry, an invalid position or fewer than 2 points. */
export function parseLineGeometry(json: string): LatLon[] | null {
  let value: unknown;
  try {
    value = JSON.parse(json);
  } catch {
    return null;
  }
  const g = value as Partial<LineGeometry> | null;
  if (!g || g.type !== 'LineString' || !Array.isArray(g.coordinates) || g.coordinates.length < 2) return null;
  const points: LatLon[] = [];
  for (const c of g.coordinates) {
    if (!Array.isArray(c) || c.length < 2) return null;
    const p = { lat: Number(c[1]), lon: Number(c[0]) };
    if (!isValidLatLon(p)) return null;
    points.push(p);
  }
  return points;
}
```

```ts
import { inverse } from '@/lib/geo/measure/geodesic';
import type { LatLon } from '@/lib/geo/types';

/** Geodesic length of each segment, in meters (empty for fewer than 2 points). */
export function segmentLengths(points: readonly LatLon[]): number[] {
  const out: number[] = [];
  for (let i = 1; i < points.length; i++) out.push(inverse(points[i - 1], points[i]).distanceM);
  return out;
}

export function totalLength(points: readonly LatLon[]): number {
  return segmentLengths(points).reduce((a, b) => a + b, 0);
}
```

```ts
export type PlaceLineInput = { missionId: string; points: LatLon[]; lineType: LineType; name: string; notes?: string | null };

function checkLinePoints(points: LatLon[]): 'invalid_coordinate' | 'too_few_points' | null {
  if (points.length < 2) return 'too_few_points';
  return points.every(isValidLatLon) ? null : 'invalid_coordinate';
}

export async function placeLine(
  ctx: CommandContext,
  input: PlaceLineInput,
): Promise<Result<{ id: string }, 'invalid_coordinate' | 'too_few_points' | 'invalid_name' | 'write_failed'>> {
  const bad = checkLinePoints(input.points);
  if (bad) return err(bad, bad === 'too_few_points' ? 'A line needs at least 2 points' : 'Coordinate is out of range');
  const name = input.name.trim();
  if (!name) return err('invalid_name', 'Name is required');
  try {
    const item = await ctx.db.write(() =>
      ctx.db.get<MapItem>(Tables.mapItems).create((i) => {
        i.missionId = input.missionId;
        i.kind = 'line';
        i.geometry = lineGeometry(input.points);
        i.sidc = null;
        i.name = name;
        i.notes = input.notes ?? null;
        i.style = lineStyle(input.lineType);
        i.createdBy = ctx.actorId();
        i.updatedBy = ctx.actorId();
      }),
    );
    return ok({ id: item.id });
  } catch (e) {
    return err('write_failed', String(e));
  }
}

export type UpdateLinePatch = { points?: LatLon[]; lineType?: LineType; name?: string; notes?: string | null };

export async function updateLine(
  ctx: CommandContext,
  id: string,
  patch: UpdateLinePatch,
): Promise<Result<void, 'not_found' | 'invalid_coordinate' | 'too_few_points' | 'invalid_name' | 'write_failed'>> {
  if (patch.points) {
    const bad = checkLinePoints(patch.points);
    if (bad) return err(bad, bad === 'too_few_points' ? 'A line needs at least 2 points' : 'Coordinate is out of range');
  }
  if (patch.name !== undefined && !patch.name.trim()) return err('invalid_name', 'Name is required');
  const item = await findItem(ctx.db, id);
  if (!item || item.kind !== 'line') return err('not_found', `No line ${id}`);
  try {
    await ctx.db.write(() =>
      item.update((i) => {
        if (patch.points) i.geometry = lineGeometry(patch.points);
        if (patch.lineType) i.style = lineStyle(patch.lineType);
        if (patch.name !== undefined) i.name = patch.name.trim();
        if (patch.notes !== undefined) i.notes = patch.notes;
        i.updatedBy = ctx.actorId();
      }),
    );
    return ok(undefined);
  } catch (e) {
    return err('write_failed', String(e));
  }
}
```

```ts
import { inverse } from '@/lib/geo/measure/geodesic';
import type { LatLon } from '@/lib/geo/types';

import { parseLineGeometry } from './geometry';
import type { MapItemSnapshot } from './repository';

type LineProps = { id: string; name: string; label: string; lineType: LineType; dash: 'solid' | 'dashed'; width: 'normal' | 'heavy' | 'wide'; endArrow: boolean; decoration: 'none' | 'flot-arcs' };
export type LineFeature = { type: 'Feature'; id: string; geometry: { type: 'LineString'; coordinates: [number, number][] }; properties: LineProps };
export type LineEndFeature = {
  type: 'Feature';
  id: string;
  geometry: { type: 'Point'; coordinates: [number, number] };
  properties: { id: string; label: string; end: 'start' | 'end'; endArrow: boolean; bearing: number };
};
export type LineFeatures = {
  lines: { type: 'FeatureCollection'; features: LineFeature[] };
  ends: { type: 'FeatureCollection'; features: LineEndFeature[] };
};

const lonLat = (p: LatLon): [number, number] => [p.lon, p.lat];
const bearing = (from: LatLon, to: LatLon) => inverse(from, to).azimuthTrue ?? 0;

function build(item: MapItemSnapshot): { line: LineFeature; ends: LineEndFeature[] } | null {
  const points = parseLineGeometry(item.geometry);
  if (!points) return null;
  const lineType = lineTypeOf(item.style);
  const info = lineTypeInfo(lineType);
  const label = lineLabel(lineType, item.name);
  const last = points.length - 1;
  const end = (which: 'start' | 'end'): LineEndFeature => ({
    type: 'Feature',
    id: `${item.id}:${which}`,
    geometry: { type: 'Point', coordinates: lonLat(which === 'start' ? points[0] : points[last]) },
    properties: {
      id: item.id,
      label,
      end: which,
      endArrow: info.endArrow,
      bearing: which === 'start' ? bearing(points[1], points[0]) : bearing(points[last - 1], points[last]),
    },
  });
  return {
    line: {
      type: 'Feature',
      id: item.id,
      geometry: { type: 'LineString', coordinates: points.map(lonLat) },
      properties: { id: item.id, name: item.name, label, lineType, dash: info.dash, width: info.width, endArrow: info.endArrow, decoration: info.decoration },
    },
    ends: [end('start'), end('end')],
  };
}

/** Saved lines for LinesLayer; features are cached per id + updatedAt (like the point selector). */
export function createLineFeatureSelector(): (items: readonly MapItemSnapshot[]) => LineFeatures {
  let cache = new Map<string, { updatedAt: number; built: { line: LineFeature; ends: LineEndFeature[] } }>();
  return (items) => {
    const next = new Map<string, { updatedAt: number; built: { line: LineFeature; ends: LineEndFeature[] } }>();
    const lines: LineFeature[] = [];
    const ends: LineEndFeature[] = [];
    for (const item of items) {
      if (item.kind !== 'line') continue;
      const cached = cache.get(item.id);
      const built = cached && cached.updatedAt === item.updatedAt ? cached.built : build(item);
      if (!built) continue;
      next.set(item.id, { updatedAt: item.updatedAt, built });
      lines.push(built.line);
      ends.push(...built.ends);
    }
    cache = next;
    return { lines: { type: 'FeatureCollection', features: lines }, ends: { type: 'FeatureCollection', features: ends } };
  };
}
```

```ts
import type { LatLon } from '@/lib/geo/types';

/** A line being edited: its points, the selected one, and where the next point is inserted. */
export type LineEdit = { vertices: LatLon[]; selected: number | null; insertAfter: number | null };

/** Appends, or inserts right after `insertAfter` (selecting it and moving the insert point along). */
export function addVertex(e: LineEdit, p: LatLon): LineEdit {
  if (e.insertAfter === null) return { ...e, vertices: [...e.vertices, p] };
  const at = e.insertAfter + 1;
  return { vertices: [...e.vertices.slice(0, at), p, ...e.vertices.slice(at)], selected: at, insertAfter: at };
}

export function moveVertex(e: LineEdit, i: number, p: LatLon): LineEdit {
  if (i < 0 || i >= e.vertices.length) return e;
  return { ...e, vertices: e.vertices.map((v, k) => (k === i ? p : v)) };
}

/** A saved shape keeps at least 2 points; while building (fewer than 2) anything goes. */
export function canRemove(e: LineEdit, i: number): boolean {
  return i >= 0 && i < e.vertices.length && (e.vertices.length > 2 || e.vertices.length < 2);
}

export function removeVertex(e: LineEdit, i: number): LineEdit {
  if (!canRemove(e, i)) return e;
  const shift = (k: number | null) => (k === null || k === i ? null : k > i ? k - 1 : k);
  return { vertices: e.vertices.filter((_, k) => k !== i), selected: shift(e.selected), insertAfter: shift(e.insertAfter) };
}

/** Tapping the selected point again deselects it; any selection ends insert mode. */
export function selectVertex(e: LineEdit, i: number | null): LineEdit {
  const selected = i !== null && i === e.selected ? null : i;
  return { ...e, selected, insertAfter: null };
}

export function startInsertAfter(e: LineEdit, i: number): LineEdit {
  return { ...e, selected: i, insertAfter: i };
}
```

```ts
import { segmentLengths } from '@/features/map-items/line-measure';
import { lineLabel, lineTypeInfo, type LineType } from '@/features/symbology/line-types';
import { formatDistance } from '@/lib/geo/format/measure';
import { midpoint } from '@/lib/geo/measure/geodesic';
import type { LatLon } from '@/lib/geo/types';

type Role = 'line' | 'segment' | 'vertex' | 'label';
type Props = { role: Role; label?: string; number?: string; index?: number; selected?: boolean; dash?: string; width?: string };
type Feature =
  | { type: 'Feature'; id: string; geometry: { type: 'LineString'; coordinates: [number, number][] }; properties: Props }
  | { type: 'Feature'; id: string; geometry: { type: 'Point'; coordinates: [number, number] }; properties: Props };

const lonLat = (p: LatLon): [number, number] => [p.lon, p.lat];

/** The line being drawn: its line, distance chips, numbered vertices and the name chip at the end. */
export function draftLineFeatures(d: { vertices: LatLon[]; selected: number | null; lineType: LineType; name: string }): { type: 'FeatureCollection'; features: Feature[] } {
  const { vertices } = d;
  const features: Feature[] = [];
  if (vertices.length >= 2) {
    const info = lineTypeInfo(d.lineType);
    features.push({ type: 'Feature', id: 'line', geometry: { type: 'LineString', coordinates: vertices.map(lonLat) }, properties: { role: 'line', dash: info.dash, width: info.width } });
    segmentLengths(vertices).forEach((m, i) =>
      features.push({
        type: 'Feature',
        id: `segment-${i}`,
        geometry: { type: 'Point', coordinates: lonLat(midpoint(vertices[i], vertices[i + 1])) },
        properties: { role: 'segment', label: formatDistance(m) },
      }),
    );
  }
  vertices.forEach((v, i) =>
    features.push({ type: 'Feature', id: `vertex-${i}`, geometry: { type: 'Point', coordinates: lonLat(v) }, properties: { role: 'vertex', number: String(i + 1), index: i, selected: d.selected === i } }),
  );
  if (vertices.length >= 2 && d.name.trim()) {
    features.push({ type: 'Feature', id: 'label', geometry: { type: 'Point', coordinates: lonLat(vertices[vertices.length - 1]) }, properties: { role: 'label', label: lineLabel(d.lineType, d.name) } });
  }
  return { type: 'FeatureCollection', features };
}
```

```ts
export type DraftState = ReturnType<typeof useDraft.getState>;

/** What differs between unit/point menus and line menus; the panels do the rest. */
export type ItemAdapter = {
  canSave(d: DraftState): boolean;
  place(ctx: CommandContext, missionId: string, d: DraftState): Promise<Result<{ id: string }, string>>;
  update(ctx: CommandContext, id: string, d: DraftState): Promise<Result<void, string>>;
  /** Loads a saved item into the draft, read-only; false when its geometry is unreadable. */
  select(item: MapItemSnapshot): boolean;
  ref(d: DraftState): ItemRef;
};

const notes = (d: DraftState) => d.notes.trim() || null;

const symbolAdapter: ItemAdapter = {
  canSave: (d) => d.coord !== null && d.coordInputValid && d.name.trim() !== '',
  place: (ctx, missionId, d) => placePoint(ctx, { missionId, coord: d.coord!, sidc: draftSidc(d), name: d.name, notes: notes(d) }),
  update: (ctx, id, d) => updateItem(ctx, id, { coord: d.coord!, name: d.name, notes: notes(d), sidc: draftSidc(d) }),
  select: (item) => {
    const at = pointCoord(item.geometry);
    if (!at) return false;
    const unit = (item.sidc ? decomposeSidc(item.sidc) : null) ?? { affiliation: 'unknown' as const, unitType: 'unspecified' as const };
    useDraft.getState().startEdit({
      itemId: item.id,
      coord: at,
      category: pointTypeForSidc(item.sidc) ? 'point' : 'unit',
      affiliation: unit.affiliation,
      unitType: unit.unitType,
      pointType: pointTypeForSidc(item.sidc) ?? 'checkpoint',
      name: item.name,
      notes: item.notes ?? '',
    });
    return true;
  },
  ref: (d) => ({ kind: 'point', sidc: draftSidc(d) }),
};

const lineAdapter: ItemAdapter = {
  canSave: (d) => d.vertices.length >= 2 && d.name.trim() !== '',
  place: (ctx, missionId, d) => placeLine(ctx, { missionId, points: d.vertices, lineType: d.lineType, name: d.name, notes: notes(d) }),
  update: (ctx, id, d) => updateLine(ctx, id, { points: d.vertices, lineType: d.lineType, name: d.name, notes: notes(d) }),
  select: (item) => {
    const points = parseLineGeometry(item.geometry);
    if (!points) return false;
    useDraft.getState().startEditLine({ itemId: item.id, points, lineType: lineTypeOf(item.style), name: item.name, notes: item.notes ?? '' });
    return true;
  },
  ref: () => ({ kind: 'line', sidc: null }),
};
```

```ts
export function observeMissionCategoryCounts(db: Database, missionId: string): Observable<{ units: number; points: number; lines: number }> {
  const items = db.get<MapItem>(Tables.mapItems);
  const inMission = Q.where('mission_id', missionId);
  return combineLatest([
    items.query(inMission, Q.where('kind', 'point')).observeCount(),
    items.query(inMission, Q.where('kind', 'point'), Q.where('sidc', Q.oneOf([...POINT_SIDCS]))).observeCount(),
    items.query(inMission, Q.where('kind', 'line')).observeCount(),
  ]).pipe(map(([symbols, points, lines]) => ({ units: Math.max(0, symbols - points), points, lines })));
}
```

```ts
/** "2 units · 1 point · 0 lines" (row) or, with ' and ', "2 units, 1 point and 0 lines" (sentence). */
export function formatCategoryCounts(c: { units: number; points: number; lines: number }, joiner = ' · '): string {
  const parts = [plural(c.units, 'unit'), plural(c.points, 'point'), plural(c.lines, 'line')];
  return joiner === ' and ' ? `${parts[0]}, ${parts[1]} and ${parts[2]}` : parts.join(joiner);
}
```
